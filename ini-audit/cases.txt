# The battery, as data rather than as code, so that the Win32 exerciser and
# the Rust implementation can be asked exactly the same questions.
#
#   name | fixture bytes | op | args...
#
# Fixture bytes take \r \n \t \0 \\ \xHH. `@empty` is an empty argument and
# `@null` a missing one; both are things a literal field cannot spell.
#
# Operations:
#   get <section> <key> <default>            GetPrivateProfileStringW
#   int <section> <key> <default>            GetPrivateProfileIntW
#   keys <section>                           ...with a null key name
#   sections                                 ...with a null section name
#   write <section> <key|@null> <val|@null>  WritePrivateProfileStringW, then
#                                            the whole file back
#   truncate <section> <key> <default> <n>   ...into a buffer of n wide chars

# --- the ordinary case, so a difference elsewhere is not the harness -------
plain                | [s]\r\nKey=value\r\n | get | s | Key | DEF
missing-key          | [s]\r\nKey=value\r\n | get | s | Other | DEF
missing-section      | [s]\r\nKey=value\r\n | get | nope | Key | DEF
empty-file           | @empty | get | s | Key | DEF

# --- quoting. MSDN says the marks are discarded; the plan assumed not ------
double-quoted        | [s]\r\nKey="value"\r\n | get | s | Key | DEF
single-quoted        | [s]\r\nKey='value'\r\n | get | s | Key | DEF
quoted-with-spaces   | [s]\r\nKey="  spaced  "\r\n | get | s | Key | DEF
quote-only-leading   | [s]\r\nKey="value\r\n | get | s | Key | DEF
quote-only-trailing  | [s]\r\nKey=value"\r\n | get | s | Key | DEF
quote-mismatched     | [s]\r\nKey="value'\r\n | get | s | Key | DEF
quote-inside         | [s]\r\nKey=a"b"c\r\n | get | s | Key | DEF
quote-empty          | [s]\r\nKey=""\r\n | get | s | Key | DEF
quote-doubled        | [s]\r\nKey=""value""\r\n | get | s | Key | DEF

# --- whitespace -----------------------------------------------------------

space-before-key     | [s]\r\n   Key=value\r\n | get | s | Key | DEF
tab-around-value     | [s]\r\nKey=\tvalue\t\r\n | get | s | Key | DEF
space-in-section     | [ s ]\r\nKey=value\r\n | get | s | Key | DEF
space-in-section-q   | [ s ]\r\nKey=value\r\n | get | @empty | Key | DEF
value-with-spaces    | [s]\r\nKey=two words\r\n | get | s | Key | DEF

# --- case, which decides whether `ColorFlag` finds `colorflag` -------------

# --- duplicates. First or last wins is not a detail: TERATERM.INI is
#     round-tripped, so the wrong answer silently changes a setting. --------
